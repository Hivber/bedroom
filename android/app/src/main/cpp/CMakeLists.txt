cmake_minimum_required(VERSION 3.22.1)
project(backrooms LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(backrooms SHARED native-lib.cpp)
target_include_directories(backrooms PRIVATE ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(native_app_glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
target_include_directories(native_app_glue PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
target_link_libraries(backrooms native_app_glue android log EGL GLESv3)
